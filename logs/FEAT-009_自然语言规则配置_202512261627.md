```javascript
async detectConflicts(newRule) {
  const existingRules = await Rule.findAll({ where: { is_active: true } });
  const conflicts = [];
  for (const existing of existingRules) {
    const duplicateConditions = this.checkDuplicateConditions(newRule, existing);
    if (duplicateConditions) {
      conflicts.push({
        conflict_type: 'duplicate',
        conflicting_rule: existing.name,
        description: `与规则"${existing.name}"存在重复条件`
      });
    }
  }
  return conflicts;
}
```
